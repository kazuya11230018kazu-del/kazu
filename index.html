var ss = SpreadsheetApp.getActiveSpreadsheet(); // プリフライトリクエストやCORSエラーを防ぐためのヘッダー付与 function sendJSON(data) { return ContentService.createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); } function doGet(e) { try { var action = e.parameter.action; if (action === "menu") { var sheet = ss.getSheetByName("メニューマスタ"); var rows = sheet.getDataRange().getValues(); var data = []; for (var i = 1; i < rows.length; i++) { var row = rows[i]; // 販売中 (E列 / インデックス4) が true のもの if (row[4] === true || row[4] === "TRUE") { data.push({ id: row[0], name: row[1], price: row[2], category: row[3] }); } } return sendJSON(data); } return sendJSON({status: "error", message: "Invalid action"}); } catch (err) { return sendJSON({status: "error", message: err.toString()}); } } function doPost(e) { try { var data = JSON.parse(e.postData.contents); var sheet = ss.getSheetByName("注文一覧"); var timestamp = new Date(); var orderId = "ORD-" + Utilities.formatDate(timestamp, "JST", "yyyyMMddHHmmss") + "-" + Math.floor(Math.random() * 900 + 100); data.items.forEach(function(item) { var subtotal = item.price * item.quantity; sheet.appendRow([ orderId, // A: 注文ID timestamp, // B: 注文日時 data.tableNo, // C: テーブルNo item.id, // D: 商品ID item.name, // E: 商品名 item.price, // F: 単価 item.quantity, // G: 数量 subtotal, // H: 小計 "未調理", // I: ステータス "" // J: 会計日時 ]); }); return sendJSON({status: "success", orderId: orderId}); } catch (error) { return sendJSON({status: "error", message: error.toString()}); } }
