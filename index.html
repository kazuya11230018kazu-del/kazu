// ========================================== // モバイルオーダー フロントエンド（テーブル番号手動選択版） // ========================================== // ページ読み込み時の処理 window.addEventListener("DOMContentLoaded", () => { // 1. ローカルストレージ（ブラウザの記憶）から前回のテーブル番号を確認 let tableNo = localStorage.getItem("mobile_order_table_no"); // テーブル番号がまだ決まっていない場合、または変更したい場合の選択処理 if (!tableNo) { tableNo = prompt("テーブル番号を入力してください（例: 1, 2, 3...）", "1"); // キャンセルされた場合や空欄の場合のデフォルト処理 if (!tableNo || tableNo.trim() === "") { tableNo = "1"; } // ブラウザに記憶させる（一度選べば、同じスマホなら次から聞かれないようにする） localStorage.setItem("mobile_order_table_no", tableNo.trim()); } // 2. 画面のテーブル番号表示を更新する // ※画面の「テーブル: 1番」などの要素IDやクラス名に合わせて書き換えてください const tableLabel = document.getElementById("table-label") || document.querySelector(".table-number"); if (tableLabel) { tableLabel.textContent = "テーブル: " + tableNo + "番"; } // グローバル変数として保持（注文時などに使用） window.currentTableNo = tableNo; // 3. メニューデータをGASから取得する関数を呼び出す fetchMenuFromGAS(tableNo); }); // （おまけ）「テーブル番号を変えたい」というボタンを置く場合の変更用関数 function changeTableNumber() { let newTableNo = prompt("新しいテーブル番号を入力してください", window.currentTableNo || "1"); if (newTableNo) { localStorage.setItem("mobile_order_table_no", newTableNo.trim()); location.reload(); // ページを再読み込みして反映 } } // 4. GASからメニューを取得する関数 function fetchMenuFromGAS(tableNo) { const GAS_URL = "https://script.google.com/macros/s/【あなたのGASのデプロイID】/exec"; fetch(`${GAS_URL}?action=getMenu&tableNo=${tableNo}`) .then(response => response.json()) .then(result => { if (result.status === "success") { console.log("メニュー取得成功, テーブル:", result.tableNo); // ここにメニューを描画する既存の処理を記述 // renderMenuList(result.data); } else { alert("メニューの取得に失敗しました: " + result.message); } }) .catch(error => { console.error("通信エラー:", error); }); } // 5. 注文を確定してGASへ送信する関数 function submitOrderToGAS(cartItems) { const GAS_URL = "https://script.google.com/macros/s/【你的GAS的部署ID】/exec"; const payloadData = { action: "submitOrder", payload: { tableNo: window.currentTableNo || "1", // 選択されたテーブル番号を送信 items: cartItems } }; fetch(GAS_URL, { method: "POST", headers: { Content-Type: "text/plain;charset=utf-8", }, body: JSON.stringify(payloadData) }) .then(response => response.json()) .then(result => { if (result.status === "success") { alert("注文が完了しました！ 受付番号: " + result.data.orderId); // 注文完了後の処理（カートを空にするなど） } else { alert("注文エラー: " + result.message); } }) .catch(error => { console.error("注文送信エラー:", error); alert("注文の送信に失敗しました。"); }); }
